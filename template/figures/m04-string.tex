  \draw[EMvec2] (P)--++({180+\aa}:1.15) node[EMlabs,left=3pt,pos=0.8]{$T_1$};

  \draw[EMcurve3] ($(P)+(180:0.6)$) arc (180:{180+\aa}:0.6);

  \node[EMlabs,text=EMjade] at ($(Q)+({0.5*\bb}:0.92)$){$\beta$};
