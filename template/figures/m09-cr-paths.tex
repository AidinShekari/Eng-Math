% The two paths of the Cauchy-Riemann derivation
\begin{tikzpicture}
  \begin{scope}% vertical: Delta z = i Delta y
    \draw[EMax] (-0.8,0)--(1.6,0) node[EMlabs,right]{$x$};
    \draw[EMax] (0,-0.5)--(0,3.2) node[EMlabs,above]{$y$};
    \node[EMptI] (a) at (0.8,0.9){};\node[EMptI] (b) at (0.8,2.3){};
    \draw[EMvec2,shorten >=3pt,shorten <=3pt] (b)--(a);
    \node[EMlab,right] at (a){$z$};\node[EMlab,right] at (b){$z+\Delta z$};
    \node[EMlab] at (1.6,-0.95){\EML{vertical path}{مسیر قائم}};
  \end{scope}
  \begin{scope}[xshift=6.2cm]% horizontal: Delta z = Delta x
    \draw[EMax] (-0.5,0)--(4.4,0) node[EMlabs,right]{$x$};
    \draw[EMax] (0,-0.5)--(0,3.2) node[EMlabs,above]{$y$};
    \node[EMptI] (a) at (1.0,1.5){};\node[EMptI] (b) at (3.0,1.5){};
    \draw[EMvec2,shorten >=3pt,shorten <=3pt] (b)--(a);
    \node[EMlab,above] at (a){$z$};\node[EMlab,above] at (b){$z+\Delta z$};
    \node[EMlab] at (2.0,-0.95){\EML{horizontal path}{مسیر افقی}};
  \end{scope}
\end{tikzpicture}
