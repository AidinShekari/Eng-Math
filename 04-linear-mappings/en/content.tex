This chapter examines some simple mappings from the domain plane (\EMim{z}) to the range plane (\EMim{w}).

\EMkeepfig{m04-translation}{5}

\EMsection{Translation}{Translation}

\EMdisp{\begin{aligned}
w=z+b&=(x+iy)+\operatorname{Re}\{b\}+i\operatorname{Im}\{b\}\\
&=\big(x+\operatorname{Re}\{b\}\big)+i\big(y+\operatorname{Im}\{b\}\big)
\end{aligned}}

\EMfigure{m04-translation}{Translation \EMim{w=z+b}: a region of the \EMim{z}-plane is shifted by
\EMim{b} without changing its shape}

\EMkeepfig{m04-rotation}{5}

\EMsection{Rotation}{Rotation}

\EMdisp{w\EMbr =az,\qquad\EMbr  |a|\EMbr =1,\qquad\EMbr  a\EMbr =e^{i\varphi},\qquad\EMbr  \varphi\EMbr =\arg\{a\}}
\EMdisp{z\EMbr =|z|e^{i\theta}\;\EMbr \Longrightarrow\;w\EMbr =|z|e^{i(\theta+\varphi)}}

\EMfigure{m04-rotation}{Rotation \EMim{w=az} with \EMim{|a|=1}: a region of the \EMim{z}-plane
is rotated about the origin by \EMim{\varphi=\arg a}}

\EMkeepfig{m04-dilation}{5}

\EMsection{Dilation or Contraction}{Dilation or Contraction}

\EMdisp{w\EMbr =az,\qquad\EMbr  a\in\mathbb{R}^+}
\EMdisp{z\EMbr =|z|e^{i\theta}\;\EMbr \Longrightarrow\;w\EMbr =a|z|e^{i\theta}}

\EMfigure{m04-dilation}{Contraction \EMim{w=az} with \EMim{0<a<1}; for \EMim{a>1} the mapping is
a dilation}

\EMsection{The General Linear Mapping}{The General Linear Mapping}

\EMdisp{w\EMbr =az+b,\qquad\EMbr  a,b\in\mathbb{C}}

\begin{enumerate}
\def\labelenumi{(\alph{enumi})}
\tightlist
\item
  a rotation by \EMim{\arg a}
\item
  a dilation or contraction by \EMim{|a|}
\item
  a translation by \EMim{b}
\end{enumerate}

\begin{EMexample}{}

For \EMim{w=(1+i)z+(1-i)}, find the image of the unit square.

\EMdisp{w(0)\EMbr =1-i,\qquad\EMbr  w(1)\EMbr =2,\qquad\EMbr  w(i)\EMbr =0,\qquad\EMbr  w(1+i)\EMbr =1+i}

\end{EMexample}
