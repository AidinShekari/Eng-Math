\def\EMtitlesize{24}
\begin{frame}{Fourier Analysis of Non-Periodic Functions}
\begin{minipage}[t]{500bp}\vspace{0pt}
\begin{uncoverenv}<2->
\EMp{To be able to use Fourier analysis for non-periodic functions, we first regard the function as periodic and then let the period tend to infinity.}
\end{uncoverenv}
\EMgap
\EMgap
\begin{uncoverenv}<4->
\EMeq{\tilde f(x) =a_0+\sum_{n=1}^{\infty}a_n\cos\frac{2\pi}Tnx+b_n\sin\frac{2\pi}Tnx}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<5->
\EMeq{a_0 =\frac1T\int_{-T/2}^{T/2}f(x)\,dx,\qquad a_n =\frac2T\int_{-T/2}^{T/2}f(x)\cos\frac{2\pi}Tnx\,dx,\qquad \\[4pt] b_n =\frac2T\int_{-T/2}^{T/2}f(x)\sin\frac{2\pi}Tnx\,dx}
\end{uncoverenv}
\end{minipage}
\EMside{76.2bp}{\uncover<3->{\EMfigure{1.100}{m03-periodic-extension}{A non-periodic function \EMim{f(x)} and its periodic extension \EMim{\tilde f(x)} with period \EMim{T}}}}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{Fourier Analysis of Non-Periodic Functions (continued)}
\begin{uncoverenv}<2->
\EMeq{\begin{aligned}
\tilde f(x)&=\left(\frac1T\int_{-T/2}^{T/2}f(t)\,dt\right)+\sum_{n=1}^{\infty}\Bigg\{\left(\frac2T\int_{-T/2}^{T/2}f(t)\cos\frac{2\pi}Tnt\,dt\right)\cos\frac{2\pi}Tnx\\
&\qquad+\left(\frac2T\int_{-T/2}^{T/2}f(t)\sin\frac{2\pi}Tnt\,dt\right)\sin\frac{2\pi}Tnx\Bigg\}
\end{aligned}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{\tilde f(x) =\left(\frac1T\int_{-T/2}^{T/2}f(t)\,dt\right)+\frac2T\sum_{n=1}^{\infty}\left\{\int_{-T/2}^{T/2}f(t)\left(\cos\frac{2\pi}Tnt\cos\frac{2\pi}Tnx+\sin\frac{2\pi}Tnt\sin\frac{2\pi}Tnx\right)dt\right\}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMeq{\tilde f(x) =\left(\frac1T\int_{-T/2}^{T/2}f(t)\,dt\right)+\frac2T\sum_{n=1}^{\infty}\left\{\int_{-T/2}^{T/2}f(t)\cos\frac{2\pi}Tn(t-x)\,dt\right\}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<5->
\EMitem{\textbullet}{\dimexpr\linewidth-26bp\relax}{To obtain the Fourier analysis of the function \EMim{f(x)} we must let \EMim{T} tend to \textbf{infinity}.}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<6->
\EMitem{\textbullet}{\dimexpr\linewidth-26bp\relax}{Suppose \EMim{f(x)} is an integrable function, so that}
\vspace{3bp}
\EMeq{\lim_{T\to\infty}\frac1T\int_{-T/2}^{T/2}f(t)\,dt =0}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{Fourier Analysis of Non-Periodic Functions (continued)}
\begin{uncoverenv}<2->
\EMp{Then}
\vspace{3bp}
\EMeq{f(x) =\lim_{T\to\infty}\frac2T\sum_{n=1}^{\infty}\left\{\int_{-T/2}^{T/2}f(t)\cos\frac{2\pi}Tn(t-x)\,dt\right\}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{\omega_n\triangleq\frac{2\pi}Tn\quad \Longrightarrow\quad \Delta\omega\triangleq\omega_n-\omega_{n-1} =\frac{2\pi}Tn-\frac{2\pi}T(n-1) =\frac{2\pi}T\quad \Longrightarrow\quad \frac2T =\frac{\Delta\omega}\pi}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMeq{f(x) =\lim_{T\to\infty}\frac1\pi\sum_{n=1}^{\infty}\left\{\int_{-T/2}^{T/2}f(t)\cos\omega_n(t-x)\,dt\right\}\Delta\omega}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<5->
\EMp{If \EMim{T\to\infty}, then \EMim{\sum_{n=1}^{\infty}\to\int_0^\infty}, \EMim{\Delta\omega\to d\omega} and \EMim{\omega_n\to\omega}:}
\vspace{3bp}
\EMeq{\begin{aligned}
f(x)&=\frac1\pi\int_0^\infty\left\{\int_{-\infty}^{\infty}f(t)\cos\omega(t-x)\,dt\right\}d\omega\\
&=\frac1\pi\int_0^\infty\left\{\int_{-\infty}^{\infty}f(t)\big(\cos\omega t\cos\omega x+\sin\omega t\sin\omega x\big)\,dt\right\}d\omega\\
&=\frac1\pi\int_0^\infty\left\{\cos\omega x\int_{-\infty}^{\infty}\big(f(t)\cos\omega t\big)\,dt+\sin\omega x\int_{-\infty}^{\infty}\big(f(t)\sin\omega t\big)\,dt\right\}d\omega
\end{aligned}}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{Fourier Analysis of Non-Periodic Functions (continued)}
\begin{uncoverenv}<2->
\EMeq{A(\omega)\triangleq\int_{-\infty}^{\infty}f(t)\cos\omega t\,dt,\qquad B(\omega)\triangleq\int_{-\infty}^{\infty}f(t)\sin\omega t\,dt\quad \Longrightarrow\quad f(x) =\frac1\pi\int_0^\infty\big\{A(\omega)\cos\omega x+B(\omega)\sin\omega x\big\}\,d\omega}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{The Fourier Integral}
\begin{EMbox}{imp}{1}{}
\begin{uncoverenv}<2->
\EMeq{f(x) =\frac1\pi\int_0^\infty\big\{A(\omega)\cos\omega x+B(\omega)\sin\omega x\big\}\,d\omega\qquad \text{(the Fourier integral; synthesis equation)}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{A(\omega)\triangleq\int_{-\infty}^{\infty}f(t)\cos\omega t\,dt\qquad \text{(the Fourier cosine transform)}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMeq{B(\omega)\triangleq\int_{-\infty}^{\infty}f(t)\sin\omega t\,dt\qquad \text{(the Fourier sine transform)}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<5->
\EMp{(the analysis equations)}
\end{uncoverenv}
\end{EMbox}
\begin{EMbox}{thm}{5}{}
\begin{uncoverenv}<6->
\EMp{A sufficient condition for the existence of the Fourier integral of a function \EMim{f(x)} is that the function be piecewise continuous, have right and left derivatives at every point, and be absolutely integrable, i.e.}
\vspace{3bp}
\EMeq{\int_{-\infty}^{\infty}|f(x)|\,dx<\infty}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<7->
\EMp{At points of discontinuity, the value of the Fourier integral equals the average of the left and right limits of \EMim{f(x)} at those points.}
\end{uncoverenv}
\end{EMbox}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{The Complex Form of the Fourier Integral (the Fourier Transform)}
\begin{uncoverenv}<2->
\EMeq{f(x) =\frac1\pi\int_0^\infty\left\{\int_{-\infty}^{\infty}f(t)\cos\omega(t-x)\,dt\right\}d\omega}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMp{The expression \EMim{\displaystyle\int_{-\infty}^{\infty}f(t)\cos\omega(t-x)\,dt} is an even function of \EMim{\omega}, hence}
\vspace{3bp}
\EMeq{f(x) =\frac1{2\pi}\int_{-\infty}^{\infty}\left\{\int_{-\infty}^{\infty}f(t)\cos\omega(t-x)\,dt\right\}d\omega}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMp{and the expression \EMim{\displaystyle\int_{-\infty}^{\infty}f(t)\sin\omega(t-x)\,dt} is an odd function of \EMim{\omega}, hence}
\vspace{3bp}
\EMeq{\frac{-i}{2\pi}\int_{-\infty}^{\infty}\left\{\int_{-\infty}^{\infty}f(t)\sin\omega(t-x)\,dt\right\}d\omega =0}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<5->
\EMeq{f(x) =\frac1{2\pi}\int_{-\infty}^{\infty}\left\{\int_{-\infty}^{\infty}f(t)\cos\omega(t-x)\,dt\right\}d\omega-\frac i{2\pi}\int_{-\infty}^{\infty}\left\{\int_{-\infty}^{\infty}f(t)\sin\omega(t-x)\,dt\right\}d\omega}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<6->
\EMeq{f(x) =\frac1{2\pi}\int_{-\infty}^{\infty}\left\{\int_{-\infty}^{\infty}f(t)\big(\cos\omega(t-x)-i\sin\omega(t-x)\big)\,dt\right\}d\omega}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{The Complex Form of the Fourier Integral (the Fourier Transform) (continued)}
\begin{uncoverenv}<2->
\EMeq{f(x) =\frac1{2\pi}\int_{-\infty}^{\infty}\left\{\int_{-\infty}^{\infty}f(t)e^{-i\omega(t-x)}\,dt\right\}d\omega =\frac1{2\pi}\int_{-\infty}^{\infty}\left\{\int_{-\infty}^{\infty}f(t)e^{-i\omega t}\,dt\right\}e^{i\omega x}\,d\omega}
\end{uncoverenv}
\EMgap
\begin{EMbox}{imp}{2}{}
\begin{uncoverenv}<3->
\EMeq{f(x) =\frac1{2\pi}\int_{-\infty}^{\infty}F(\omega)e^{i\omega x}\,d\omega\qquad \text{(the Fourier integral; synthesis equation)}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMeq{F(\omega) =\int_{-\infty}^{\infty}f(t)e^{-i\omega t}\,dt\qquad \text{(the Fourier transform; analysis equation)}}
\end{uncoverenv}
\end{EMbox}
\begin{uncoverenv}<5->
\EMp{The complex form of the Fourier series (review):}
\vspace{3bp}
\EMeq{f(x) =\sum_{n=-\infty}^{\infty}c_ne^{i\frac{2\pi}Tnx}\quad \text{(Fourier series; synthesis equation)},\qquad c_n =\frac1T\int_Tf(x)e^{-i\frac{2\pi}Tnx}\,dx\quad \\[4pt] \text{(Fourier coefficients; analysis equation)}}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{The Function \EMim{\operatorname{Si}(u)}}
\begin{minipage}[t]{500bp}\vspace{0pt}
\begin{uncoverenv}<2->
\EMeq{\operatorname{Si}(x)\triangleq\int_0^x\frac{\sin w}{w}\,dw}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMitem{\textbullet}{\dimexpr\linewidth-26bp\relax}{odd;}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMitem{\textbullet}{\dimexpr\linewidth-26bp\relax}{has finite asymptotic values at infinity.}
\end{uncoverenv}
\EMgap
\EMgap
\begin{EMbox}{thm}{5}{}
\begin{uncoverenv}<6->
\EMeq{\int_0^\infty\frac{\sin\alpha w}{w}\,dw =\begin{cases}\dfrac\pi2&\alpha>0\\\noalign{\vskip 1mm} 0&\alpha=0\\\noalign{\vskip 1mm} -\dfrac\pi2&\alpha<0\end{cases}}
\end{uncoverenv}
\end{EMbox}
\begin{EMbox}{prf}{6}{}
\begin{uncoverenv}<7->
\EMp{For \EMim{\alpha=0} the lemma is obvious. With the substitution \EMim{\tau=\alpha w}:}
\vspace{3bp}
\EMeq{\int_0^x\frac{\sin\alpha w}{w}\,dw =\int_0^{\alpha x}\frac{\sin\tau}{\frac\tau\alpha}\,\frac{d\tau}{\alpha} =\int_0^{\alpha x}\frac{\sin\tau}{\tau}\,d\tau =\operatorname{Si}(\alpha x),\qquad \\[4pt] \operatorname{Si}(\alpha x) =-\operatorname{Si}(-\alpha x)}
\end{uncoverenv}
\end{EMbox}
\end{minipage}
\EMside{119.3bp}{\uncover<5->{\EMfigure{1.159}{m03-si}{The function \EMim{\operatorname{Si}(u)} (solid) and the function \EMim{\dfrac{\sin u}{u}} (dashed)}}}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{The Function \EMim{\operatorname{Si}(u)} (continued)}
\begin{EMbox}{prf}{1}{}
\begin{uncoverenv}<2->
\EMp{So it suffices to prove the lemma for positive values of \EMim{\alpha}.}
\vspace{3bp}
\EMeq{F(\alpha)\triangleq\int_0^\infty e^{-\alpha t}\frac{\sin t}{t}\,dt,\qquad F(0) =\operatorname{Si}(+\infty) =\;?}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{F'(\alpha) =\int_0^\infty-t\,e^{-\alpha t}\frac{\sin t}t\,dt =-\int_0^\infty e^{-\alpha t}\sin t\,dt =-\left[\frac{e^{-\alpha t}}{1+\alpha^2}\big(-\alpha\sin t-\cos t\big)\right]_0^\infty =\frac{-1}{1+\alpha^2}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMeq{\alpha>0:\quad F(\alpha) =-\tan^{-1}\alpha+C,\qquad F(\alpha\to\infty) =0\quad \Longrightarrow\quad C =F(\infty)+\tan^{-1}\infty =\frac\pi2}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<5->
\EMeq{F(\alpha) =-\tan^{-1}\alpha+\frac\pi2\quad \Longrightarrow\quad F(0) =\operatorname{Si}(+\infty) =\frac\pi2}
\end{uncoverenv}
\end{EMbox}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{Time Scaling}
\begin{uncoverenv}<2->
\EMeq{f(x)\to f(ax),\quad a\in\mathbb{R}\quad \Longrightarrow\quad \begin{cases}A(\omega)\to\dfrac1{|a|}A\!\left(\dfrac\omega a\right)\\\noalign{\vskip 2mm} B(\omega)\to\dfrac1{|a|}B\!\left(\dfrac\omega a\right)\end{cases}}
\end{uncoverenv}
\EMgap
\begin{EMbox}{prf}{2}{}
\begin{uncoverenv}<3->

\EMgap
\begin{EMbox}{prf}{2}{}
