\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{Elementary Complex Functions}
\begin{uncoverenv}<2->
\EMp{The elementary complex functions generalize the elementary real functions.}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{The Exponential Function}
\begin{uncoverenv}<2->
\EMp{The real exponential function \EMim{f(x)=e^x} has the following properties:}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{\begin{aligned}
&1)\quad f(x_1+x_2)=f(x_1)f(x_2)\qquad &&e^{x_1}e^{x_2}=e^{x_1+x_2}\\
&2)\quad \frac{d}{dx}f(x)=f(x)\qquad &&\frac{d}{dx}e^x=e^x\\
&3)\quad e^x=1+x+\frac{x^2}{2!}+\cdots+\frac{x^n}{n!}+\cdots
\end{aligned}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMp{We define the complex exponential function inspired by the third property, and then verify the other properties for it.}
\end{uncoverenv}
\EMgap
\begin{EMbox}{def}{4}{}
\begin{uncoverenv}<5->
\EMeq{w =e^z\triangleq1+z+\frac{z^2}{2!}+\cdots+\frac{z^n}{n!}+\cdots}
\end{uncoverenv}
\end{EMbox}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{The Exponential Function (continued)}
\begin{uncoverenv}<2->
\EMp{If in the definition of the complex exponential we substitute \EMim{z=iy}:}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{\begin{aligned}
e^{iy}&=1+iy+\frac1{2!}(iy)^2+\frac1{3!}(iy)^3+\cdots\\
&=1+iy-\frac1{2!}y^2-\frac{i}{3!}y^3+\frac1{4!}y^4+\frac{i}{5!}y^5+\cdots\\
&=\underbrace{\left(1-\frac1{2!}y^2+\frac1{4!}y^4+\cdots\right)}_{\cos y}+i\underbrace{\left(y-\frac1{3!}y^3+\frac1{5!}y^5+\cdots\right)}_{\sin y}=\cos y+i\sin y
\end{aligned}}
\end{uncoverenv}
\EMgap
\begin{EMbox}{imp}{3}{}
\begin{uncoverenv}<4->
\EMeq{w =e^z =e^{x+iy} =e^xe^{iy} =e^x(\cos y+i\sin y)}
\end{uncoverenv}
\end{EMbox}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{The Exponential Function (continued)}
\begin{EMbox}{ex}{1}{}
\EMp{Is the function \EMim{w=e^z} analytic?}
\EMgap
\begin{uncoverenv}<2->
\EMeq{u(x,y) =e^x\cos y,\qquad v(x,y) =e^x\sin y}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{\frac{\partial u}{\partial x} =e^x\cos y =\frac{\partial v}{\partial y}\ \EMcheck \qquad \qquad \frac{\partial u}{\partial y} =-e^x\sin y =-\frac{\partial v}{\partial x}\ \EMcheck}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMp{Since the Cauchy--Riemann conditions hold in the whole complex plane, the function \EMim{w=e^z} is differentiable in the whole complex plane.}
\end{uncoverenv}
\end{EMbox}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{The Exponential Function (continued)}
\begin{minipage}[t]{500bp}\vspace{0pt}
\begin{EMbox}{ex}{1}{}
\EMp{Find the derivative of \EMim{w=e^z}.}
\EMgap
\begin{uncoverenv}<2->
\EMeq{\frac{d}{dz}e^z =\frac{\partial u}{\partial x}+i\,\frac{\partial v}{\partial x} =e^x\cos y+ie^x\sin y \\[4pt] =e^x(\cos y+i\sin y) =e^z}
\end{uncoverenv}
\end{EMbox}
\begin{EMbox}{imp}{2}{}
\begin{uncoverenv}<3->
\EMp{From now on we can write every complex number in exponential form:}
\vspace{3bp}
\EMeq{z =x+iy =|z|(\cos\theta+i\sin\theta) =|z|\,e^{i\theta}}
\end{uncoverenv}
\end{EMbox}
\begin{uncoverenv}<4->
\EMp{Observe that}
\vspace{3bp}
\EMeq{e^{z+i2k\pi} =e^ze^{i2k\pi} =e^z\big[\cos(2k\pi)+i\sin(2k\pi)\big] =e^z}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<5->
\EMp{so \EMim{w=e^z} is a periodic function in the complex plane with period \EMim{i2\pi}.}
\end{uncoverenv}
\EMgap
\end{minipage}
\EMside{208.0bp}{\uncover<6->{\EMfigure{1.500}{m10-exp-strip}{The fundamental strip of the exponential function: \EMim{-\pi<y\le\pi}}}}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{Trigonometric Functions}
\begin{uncoverenv}<2->
\EMeq{e^{i\theta} =\cos\theta+i\sin\theta\quad \Longrightarrow\quad e^{-i\theta} =\cos\theta-i\sin\theta}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{\Longrightarrow\quad \cos\theta =\frac{e^{i\theta}+e^{-i\theta}}{2},\qquad \sin\theta =\frac{e^{i\theta}-e^{-i\theta}}{2i}}
\end{uncoverenv}
\EMgap
\begin{EMbox}{def}{3}{}
\begin{uncoverenv}<4->
\EMeq{\cos z\triangleq\frac{e^{iz}+e^{-iz}}{2},\qquad \sin z\triangleq\frac{e^{iz}-e^{-iz}}{2i}}
\end{uncoverenv}
\end{EMbox}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{Trigonometric Functions (continued)}
\begin{uncoverenv}<2->
\EMeq{\begin{aligned}
\cos z&=\frac12\Big[e^{i(x+iy)}+e^{-i(x+iy)}\Big]=\frac12\Big[e^{ix}e^{-y}+e^{-ix}e^{y}\Big]\\
&=\frac12\Big[e^{-y}(\cos x+i\sin x)+e^{y}(\cos x-i\sin x)\Big]\\
&=\cos x\left(\frac{e^y+e^{-y}}{2}\right)-i\sin x\left(\frac{e^y-e^{-y}}{2}\right)\\
&=\cos x\cosh y-i\sin x\sinh y
\end{aligned}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{\begin{aligned}
\sin z&=\frac1{2i}\Big[e^{i(x+iy)}-e^{-i(x+iy)}\Big]=\frac1{2i}\Big[e^{ix}e^{-y}-e^{-ix}e^{y}\Big]\\
&=\frac1{2i}\Big[e^{-y}(\cos x+i\sin x)-e^{y}(\cos x-i\sin x)\Big]\\
&=\sin x\left(\frac{e^y+e^{-y}}{2}\right)+i\cos x\left(\frac{e^y-e^{-y}}{2}\right)\\
&=\sin x\cosh y+i\cos x\sinh y
\end{aligned}}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{Analyticity and Differentiability of \EMim{\cos z}}
\begin{uncoverenv}<2->
\EMeq{\cos z =u(x,y)+iv(x,y) =\cos x\cosh y-i\sin x\sinh y,\qquad u =\cos x\cosh y,\quad v =-\sin x\sinh y}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{\frac{\partial u}{\partial x} =-\sin x\cosh y =\frac{\partial v}{\partial y}\ \EMcheck \qquad \qquad \frac{\partial u}{\partial y} =\cos x\sinh y =-\frac{\partial v}{\partial x}\ \EMcheck}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMp{So this function is analytic in the whole complex plane. We obtain its derivative with one of the methods described earlier:}
\vspace{3bp}
\EMeq{f'(z) =\frac{d}{dz}\cos z =\frac{\partial u}{\partial x}+i\,\frac{\partial v}{\partial x} =-\sin x\cosh y-i\cos x\sinh y =-\sin z}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{Analyticity and Differentiability of \EMim{\sin z}}
\begin{uncoverenv}<2->
\EMeq{\sin z =u(x,y)+iv(x,y) =\sin x\cosh y+i\cos x\sinh y,\qquad u =\sin x\cosh y,\quad v =\cos x\sinh y}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{\frac{\partial u}{\partial x} =\cos x\cosh y =\frac{\partial v}{\partial y}\ \EMcheck \qquad \qquad \frac{\partial u}{\partial y} =\sin x\sinh y =-\frac{\partial v}{\partial x}\ \EMcheck}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMp{So this function is analytic in the whole complex plane and}
\vspace{3bp}
\EMeq{f'(z) =\frac{d}{dz}\sin z =\frac{\partial u}{\partial x}+i\,\frac{\partial v}{\partial x} =\cos x\cosh y-i\sin x\sinh y =\cos z}
\end{uncoverenv}
\EMgap
\begin{EMbox}{imp}{4}{}
\begin{uncoverenv}<5->
\EMeq{(\sin z)' =\cos z,\qquad (\cos z)' =-\sin z}
\end{uncoverenv}
\end{EMbox}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{Unboundedness of \EMim{\sin z} and \EMim{\cos z}}
\begin{uncoverenv}<2->
\EMeq{|\cos z| =\sqrt{\cos^2x\cosh^2y+\sin^2x\sinh^2y} =\sqrt{\cos^2x+\sinh^2y},\qquad (\cosh^2y =1+\sinh^2y,\ \ \sin^2x \\[4pt] =1-\cos^2x)}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{|\sin z| =\sqrt{\sin^2x\cosh^2y+\cos^2x\sinh^2y} =\sqrt{\sin^2x+\sinh^2y},\qquad (\cosh^2y =1+\sinh^2y,\ \ \cos^2x \\[4pt] =1-\sin^2x)}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMp{Since \EMim{\sinh y} is an unbounded function, the functions \EMim{\cos z} and \EMim{\sin z} are \textbf{unbounded}.}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{Periodicity}
\begin{uncoverenv}<2->
\EMeq{\begin{aligned}
\cos(z+2\pi)&=\cos(x+iy+2\pi)=\cos(x+2\pi)\cosh y-i\sin(x+2\pi)\sinh y\\
&=\cos x\cosh y-i\sin x\sinh y=\cos z
\end{aligned}}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{Periodicity (continued)}
\begin{minipage}[t]{500bp}\vspace{0pt}
\begin{uncoverenv}<2->
\EMp{In the same way \EMim{\sin(z+2\pi)=\sin z}.}
\end{uncoverenv}
\EMgap
\end{minipage}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{Remark}
\begin{uncoverenv}<2->
\EMeq{\cos(iy) =\cosh y,\qquad \sin(iy) =i\sinh y}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{\cos(x+iy) =\cos x\cos(iy)-\sin x\sin(iy) =\cos x\cosh y-i\sin x\sinh y}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMeq{\sin(x+iy) =\sin x\cos(iy)+\cos x\sin(iy) =\sin x\cosh y+i\cos x\sinh y}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<5->
\EMp{The other trigonometric functions:}
\vspace{3bp}
\EMeq{\tan z\triangleq\frac{\sin z}{\cos z},\qquad \cot z\triangleq\frac{\cos z}{\sin z},\qquad \sec z\triangleq\frac1{\cos z},\qquad \csc z\triangleq\frac1{\sin z}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<6->
\EMeq{(\tan z)' =\sec^2z,\quad \dots}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{Hyperbolic Functions}
\begin{uncoverenv}<2->
\EMeq{\cosh z\triangleq\frac{e^z+e^{-z}}{2},\qquad \sinh z\triangleq\frac{e^z-e^{-z}}{2},\qquad e^z =e^x(\cos y+i\sin y)}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{Hyperbolic Functions (continued)}
\begin{EMbox}{exe}{1}{}
\EMp{Verify that}
\vspace{3bp}
\EMeq{\cosh z =\cosh x\cos y+i\sinh x\sin y,\qquad \sinh z =\sinh x\cos y+i\cosh x\sin y}
\end{EMbox}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{Hyperbolic Functions (continued)}
\begin{minipage}[t]{500bp}\vspace{0pt}
\begin{EMbox}{exe}{1}{}
\EMp{Verify that \EMim{\cosh z} and \EMim{\sinh z} are analytic in the whole complex plane and that their derivatives are given by}
\vspace{3bp}
\EMeq{(\cosh z)' =\sinh z,\qquad (\sinh z)' =\cosh z}
\end{EMbox}
\begin{uncoverenv}<2->
\EMeq{\tanh z =\frac{\sinh z}{\cosh z},\quad \coth z =\frac{\cosh z}{\sinh z},\quad \operatorname{sech}z =\frac1{\cosh z},\quad \\[4pt] \operatorname{csch}z =\frac1{\sinh z}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{\cosh(z+i2\pi) =\cosh(z),\qquad \sinh(z+i2\pi) =\sinh(z)}
\end{uncoverenv}
\EMgap
\end{minipage}
\EMside{208.0bp}{\uncover<4->{\EMfigure{1.500}{m10-hyp-strip}{The fundamental strip of the hyperbolic functions: \EMim{-\pi<y\le\pi}}}}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{The Logarithm Function}
\begin{uncoverenv}<2->
\EMeq{w =\ln z,\qquad z =|z|e^{i\arg z} =e^w =e^{\ln|z|}e^{i\arg z}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{w =\ln z =\ln|z|+i\arg z =\ln\!\big(x^2+y^2\big)^{1/2}+i\tan^{-1}\frac yx}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMeq{u(x,y) =\frac12\ln\!\big(x^2+y^2\big),\qquad v(x,y) =\tan^{-1}\frac yx}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<5->
\EMeq{\frac{\partial u}{\partial x} =\frac{x}{x^2+y^2} =\frac{\partial v}{\partial y}\ \EMcheck \qquad \qquad \frac{\partial u}{\partial y} =\frac{y}{x^2+y^2} =-\frac{\partial v}{\partial x}\ \EMcheck}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<6->
\EMp{Hence the logarithm function is analytic in the whole complex plane except at the origin (a single-valued branch, such as the principal one, is analytic off its cut, here the negative real axis).}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<7->
\EMeq{\frac{d}{dz}\ln z =\frac{\partial u}{\partial x}+i\,\frac{\partial v}{\partial x} =\frac{x}{x^2+y^2}+i\,\frac{-y}{x^2+y^2} =\frac{\bar z}{z\bar z} =\frac1z}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<8->
\EMeq{z =re^{i\theta} =re^{i(\theta+2k\pi)}}
\end{uncoverenv}
\end{frame}
