\begin{uncoverenv}<2->
\EMp{در این فصل برخی نگاشت‌های ساده از صفحۀ دامنه (\EMim{z}) به صفحۀ برد (\EMim{w}) بررسی می‌شوند.}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{انتقال}
\begin{minipage}[t]{500bp}\vspace{0pt}
\begin{uncoverenv}<2->
\EMeq{\begin{aligned}
w=z+b&=(x+iy)+\operatorname{Re}\{b\}+i\operatorname{Im}\{b\}\\
&=\big(x+\operatorname{Re}\{b\}\big)+i\big(y+\operatorname{Im}\{b\}\big)
\end{aligned}}
\end{uncoverenv}
\EMgap
\end{minipage}
\EMside{77.0bp}{\uncover<3->{\EMfigure{0.990}{m11-translation}{انتقال \EMim{w=z+b}: ناحیه‌ای از صفحۀ \EMim{z} بدون تغییر شکل به اندازۀ \EMim{b} جابه‌جا می‌شود}}}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{دوران}
\begin{minipage}[t]{500bp}\vspace{0pt}
\begin{uncoverenv}<2->
\EMeq{w =az,\qquad |a| =1,\qquad a =e^{i\varphi},\qquad \varphi =\arg\{a\}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{z =|z|e^{i\theta}\; \Longrightarrow\;w =|z|e^{i(\theta+\varphi)}}
\end{uncoverenv}
\EMgap
\end{minipage}
\EMside{86.9bp}{\uncover<4->{\EMfigure{0.990}{m11-rotation}{دوران \EMim{w=az} با \EMim{|a|=1}: ناحیه‌ای از صفحۀ \EMim{z} به اندازۀ \EMim{\varphi=\arg a} حول مبدأ می‌چرخد}}}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{انبساط یا انقباض}
\begin{minipage}[t]{500bp}\vspace{0pt}
\begin{uncoverenv}<2->
\EMeq{w =az,\qquad a\in\mathbb{R}^+}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{z =|z|e^{i\theta}\; \Longrightarrow\;w =a|z|e^{i\theta}}
\end{uncoverenv}
\EMgap
\end{minipage}
\EMside{86.9bp}{\uncover<4->{\EMfigure{0.990}{m11-dilation}{انقباض \EMim{w=az} با \EMim{0<a<1}؛ برای \EMim{a>1} نگاشت یک انبساط است}}}
\end{frame}

\def\EMtitlesize{26}\def\EMtitleskip{31}
\begin{frame}{نگاشت خطی کلی}
\begin{uncoverenv}<2->
\EMeq{w =az+b,\qquad a,b\in\mathbb{C}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMp{الف) دوران به اندازۀ \EMim{\arg a}
ب) انبساط یا انقباض به اندازۀ \EMim{|a|}
ج) انتقال به اندازۀ \EMim{b}}
\end{uncoverenv}
\end{frame}
