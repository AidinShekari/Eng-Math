\begin{frame}{نگاشت‌های خطی}
\begin{uncoverenv}<2->
\EMp{در این فصل برخی نگاشت‌های ساده از صفحهٔ دامنه (\EMim{z}) به صفحهٔ برد (\EMim{w}) بررسی می‌شوند.}
\end{uncoverenv}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{انتقال}
\begin{minipage}[t]{500bp}\vspace{0pt}
\begin{uncoverenv}<2->
\EMeq{\begin{aligned}
w=z+b&=(x+iy)+\operatorname{Re}\{b\}+i\operatorname{Im}\{b\}\\
&=\big(x+\operatorname{Re}\{b\}\big)+i\big(y+\operatorname{Im}\{b\}\big)
\end{aligned}}
\end{uncoverenv}
\EMgap
\end{minipage}
\EMside{69.8bp}{\uncover<3->{\EMfigure{0.988}{m11-translation}{انتقال \EMim{w=z+b}: ناحیه‌ای از صفحهٔ \EMim{z} بدون تغییر شکل به اندازهٔ \EMim{b} جابه‌جا می‌شود}}}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{دوران}
\begin{minipage}[t]{500bp}\vspace{0pt}
\begin{uncoverenv}<2->
\EMeq{w =az,\qquad |a| =1,\qquad a =e^{i\varphi},\qquad \varphi =\arg\{a\}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{z =|z|e^{i\theta}\; \Longrightarrow\;w =|z|e^{i(\theta+\varphi)}}
\end{uncoverenv}
\EMgap
\end{minipage}
\EMside{79.7bp}{\uncover<4->{\EMfigure{0.988}{m11-rotation}{دوران \EMim{w=az} با \EMim{|a|=1}: ناحیه‌ای از صفحهٔ \EMim{z} به اندازهٔ \EMim{\varphi=\arg a} حول مبدأ می‌چرخد}}}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{انبساط یا انقباض}
\begin{minipage}[t]{500bp}\vspace{0pt}
\begin{uncoverenv}<2->
\EMeq{w =az,\qquad a\in\mathbb{R}^+}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMeq{z =|z|e^{i\theta}\; \Longrightarrow\;w =a|z|e^{i\theta}}
\end{uncoverenv}
\EMgap
\end{minipage}
\EMside{79.7bp}{\uncover<4->{\EMfigure{0.988}{m11-dilation}{انقباض \EMim{w=az} با \EMim{0<a<1}؛ برای \EMim{a>1} نگاشت یک انبساط است}}}
\end{frame}

\def\EMtitlesize{24}
\begin{frame}{نگاشت خطی کلی}
\begin{uncoverenv}<2->
\EMeq{w =az+b,\qquad a,b\in\mathbb{C}}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<3->
\EMp{الف) دوران به اندازهٔ \EMim{\arg a}
ب) انبساط یا انقباض به اندازهٔ \EMim{|a|}
ج) انتقال به اندازهٔ \EMim{b}}
\end{uncoverenv}
\end{frame}
