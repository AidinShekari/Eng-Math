\end{uncoverenv}

\end{uncoverenv}
\EMgap
\begin{uncoverenv}<5->
\EMp{(روابط آنالیز)}
\end{uncoverenv}
\end{EMbox}
\begin{EMbox}{thm}{5}{}
\begin{uncoverenv}<6->

\vspace{3bp}
\EMeq{\int_{-\infty}^{\infty}|f(x)|\,dx<\infty}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<7->
\EMp{مقدار انتگرال فوریه در نقاط ناپیوستگی برابر است با متوسط حد چپ و راست تابع \EMim{f(x)} در آن نقاط.}
\end{uncoverenv}
\end{EMbox}
\end{frame}

\EMgap
\begin{EMbox}{thm}{5}{}
\begin{uncoverenv}<6->
\EMeq{\int_0^\infty\frac{\sin\alpha w}{w}\,dw =\begin{cases}\dfrac\pi2&\alpha>0\\\noalign{\vskip 1mm} 0&\alpha=0\\\noalign{\vskip 1mm} -\dfrac\pi2&\alpha<0\end{cases}}
\end{uncoverenv}
\end{EMbox}
\begin{EMbox}{prf}{6}{}
\begin{uncoverenv}<7->
\EMp{برای \EMim{\alpha=0} درستی لم واضح است. با تغییر متغیر \EMim{\tau=\alpha w}:}
\vspace{3bp}
\EMeq{\int_0^x\frac{\sin\alpha w}{w}\,dw =\int_0^{\alpha x}\frac{\sin\tau}{\frac\tau\alpha}\,\frac{d\tau}{\alpha} =\int_0^{\alpha x}\frac{\sin\tau}{\tau}\,d\tau =\operatorname{Si}(\alpha x),\qquad \\[4pt] \operatorname{Si}(\alpha x) =-\operatorname{Si}(-\alpha x)}
\end{uncoverenv}
\end{EMbox}

\EMgap
\begin{uncoverenv}<2->

\vspace{3bp}
\EMeq{B(\omega) =\int_{-a}^{a}x^2\sin\omega x\,dx =0}

\EMgap
\begin{uncoverenv}<3->
\EMeq{A(\omega) =\int_{-\infty}^{\infty}f(x)\cos\omega x\,dx =2\int_0^ax^2\cos\omega x\,dx =2\left[\frac{x^2}\omega\sin\omega x+\frac{2x}{\omega^2}\cos\omega x-\frac2{\omega^3}\sin\omega x\right]_0^a \\[4pt] =2\left[\left(\frac{a^2}\omega-\frac2{\omega^3}\right)\sin\omega a+\frac{2a}{\omega^2}\cos\omega a\right]}
\end{uncoverenv}
\EMgap
\begin{uncoverenv}<4->
\EMeq{f(x) =\frac2\pi\int_0^\infty\left[\left(\frac{a^2}\omega-\frac2{\omega^3}\right)\sin\omega a+\frac{2a}{\omega^2}\cos\omega a\right]\cos\omega x\,d\omega}
